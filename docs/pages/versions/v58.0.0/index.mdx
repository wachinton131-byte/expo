
import React, { useState } from 'react';
import {
  SafeAreaView,
  View,
  Text,
  TouchableOpacity,
  StyleSheet
} from 'react-native';

export default function App() {
  const [debut, setDebut] = useState(null);
  const [fin, setFin] = useState(null);

  const heures = debut && fin
    ? ((fin.getTime() - debut.getTime()) / 3600000).toFixed(2)
    : '0.00';

  function heure(date) {
    return date
      ? date.toLocaleTimeString('fr-FR', {
          hour: '2-digit',
          minute: '2-digit'
        })
      : '--:--';
  }

  return (
    <SafeAreaView style={styles.container}>
      <Text style={styles.logo}>Mes Pointages</Text>
      <Text style={styles.sousTitre}>
        Suivi de tes heures de travail
      </Text>

      <View style={styles.card}>
        <Text style={styles.titre}>Mon pointage du jour</Text>

        <Text style={styles.label}>Début du travail</Text>
        <Text style={styles.heure}>{heure(debut)}</Text>

        <Text style={styles.label}>Fin du travail</Text>
        <Text style={styles.heure}>{heure(fin)}</Text>

        <TouchableOpacity
          style={styles.bouton}
          onPress={() => {
            setDebut(new Date());
            setFin(null);
          }}
        >
          <Text style={styles.boutonTexte}>COMMENCER</Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={[styles.bouton, styles.boutonFin]}
          onPress={() => debut && !fin && setFin(new Date())}
          disabled={!debut || !!fin}
        >
          <Text style={styles.boutonTexte}>TERMINER</Text>
        </TouchableOpacity>

        <View style={styles.total}>
          <Text style={styles.label}>Temps écoulé</Text>
          <Text style={styles.totalHeures}>{heures} h</Text>
        </View>
      </View>

      <Text style={styles.bas}>
        Ton temps de travail, au même endroit.
      </Text>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: '#F3F7FC',
    padding: 22,
    justifyContent: 'center'
  },
  logo: {
    fontSize: 30,
    fontWeight: 'bold',
    color: '#1769C2',
    textAlign: 'center'
  },
  sousTitre: {
    textAlign: 'center',
    color: '#667085',
    marginTop: 8,
    marginBottom: 30
  },
  card: {
    backgroundColor: 'white',
    padding: 22,
    borderRadius: 18
  },
  titre: {
    fontSize: 21,
    fontWeight: 'bold',
    marginBottom: 25,
    color: '#172B4D'
  },
  label: {
    color: '#667085',
    marginTop: 12
  },
  heure: {
    fontSize: 25,
    fontWeight: 'bold',
    color: '#172B4D'
  },
  bouton: {
    backgroundColor: '#1769C2',
    padding: 16,
    borderRadius: 12,
    marginTop: 25,
    alignItems: 'center'
  },
  boutonFin: {
    backgroundColor: '#263B53',
    marginTop: 12
  },
  boutonTexte: {
    color: 'white',
    fontWeight: 'bold'
  },
  total: {
    marginTop: 25,
    padding: 16,
    backgroundColor: '#EAF2FC',
    borderRadius: 12
  },
  totalHeures: {
    fontSize: 30,
    fontWeight: 'bold',
    color: '#1769C2'
  },
  bas: {
    textAlign: 'center',
    color: '#667085',
    marginTop: 25
  }
});
